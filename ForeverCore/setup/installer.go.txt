// Windows one-click updater. The CI copies the built ZIP to installer_payload.zip
// beside this source and compiles it into a single setup EXE (not included in Go unit builds).
package main

import (
 "archive/zip"
 "bytes"
 "encoding/json"
 "fmt"
 "io"
 "os"
 "os/exec"
 "path/filepath"
 "strings"
 "bufio"
 _ "embed"
)

//go:embed installer_payload.zip
var archive []byte

func waitAndFail(err error) {
 fmt.Fprintln(os.Stderr,"ForeverCore Installation fehlgeschlagen:",err)
 fmt.Println("Bitte Fehlermeldung sichern. Enter zum Schliessen.")
 _,_=bufio.NewReader(os.Stdin).ReadString('\n')
 os.Exit(1)
}
func installRoot() string {
 home,_:=os.UserHomeDir()
 desktop:=filepath.Join(home,"Desktop")
 if _,err:=os.Stat(desktop);err!=nil { desktop=home }
 return filepath.Join(desktop,"ForeverCore_0.6.0_AllesInEinem")
}
func lastAddons(root string) string {
 if val:=os.Getenv("WOW_ADDONS_PATH");val!="" { if isDir(val){return val} }
 old:=[]string{
  root,
  filepath.Join(filepath.Dir(root),"ForeverCore_0.6.0_Integration_Test"),
  filepath.Join(filepath.Dir(root),"ForeverCore_0.6.0_Windows_Test"),
 }
 for _,dir:=range old {
  b,err:=os.ReadFile(filepath.Join(dir,"config.json"));if err!=nil{continue}
  var cfg struct{Addons string `json:"addons"`}
  if json.Unmarshal(b,&cfg)==nil && isDir(cfg.Addons){return cfg.Addons}
 }
 home,_:=os.UserHomeDir()
 candidates:=[]string{
  `C:\Program Files (x86)\World of Warcraft\_retail_\Interface\AddOns`,
  `C:\Program Files (x86)\World of Warcraft\_classic_\Interface\AddOns`,
  `C:\Program Files\World of Warcraft\_retail_\Interface\AddOns`,
  filepath.Join(home,"Games","World of Warcraft","Interface","AddOns"),
 }
 for _,path:=range candidates{if isDir(path){return path}}
 return ""
}
func isDir(path string) bool { s,e:=os.Stat(path);return e==nil && s.IsDir() }
func chooseAddons(root string) string {
 val:=lastAddons(root)
 reader:=bufio.NewReader(os.Stdin)
 for {
  if val=="" {fmt.Println("Bitte den WoW-Ordner Interface\\AddOns eingeben (oder hierher ziehen):")}
  if val!="" {fmt.Println("WoW-Addon-Ordner:",val);fmt.Print("Enter = verwenden; oder anderen Pfad eingeben: ")}
  raw,e:=reader.ReadString('\n');if e!=nil&&len(raw)==0 {waitAndFail(e)}
  raw=strings.Trim(strings.TrimSpace(raw),"\"")
  if raw!=""{val=raw}
  if isDir(val){return val}
  fmt.Println("Ordner nicht gefunden. Noch einmal pruefen.")
  val=""
 }
}
func unpack(target string) error {
 z,err:=zip.NewReader(bytes.NewReader(archive),int64(len(archive)))
 if err!=nil{return err}
 for _,entry:=range z.File{
  name:=strings.ReplaceAll(entry.Name,"\\","/")
  parts:=strings.SplitN(name,"/",2)
  if len(parts)<2 || parts[1]=="" {continue}
  rel:=filepath.FromSlash(parts[1])
  if filepath.IsAbs(rel) || rel==".." || strings.HasPrefix(rel,".."+string(filepath.Separator)){return fmt.Errorf("Unerlaubter Dateipfad")}
  dest:=filepath.Join(target,rel)
  if entry.FileInfo().IsDir(){if err:=os.MkdirAll(dest,0755);err!=nil{return err};continue}
  if err:=os.MkdirAll(filepath.Dir(dest),0755);err!=nil{return err}
  in,err:=entry.Open();if err!=nil{return err}
  out,err:=os.OpenFile(dest,os.O_WRONLY|os.O_CREATE|os.O_TRUNC,0644)
  if err!=nil{in.Close();return err}
  _,copyErr:=io.Copy(out,in);closeErr:=out.Close();in.Close()
  if copyErr!=nil{return copyErr};if closeErr!=nil{return closeErr}
 }
 return nil
}
func main(){
 fmt.Println("ForeverCore Komplett-Update")
 fmt.Println("Bitte WoW und eine bisher laufende ForeverCore.exe zuerst schliessen.")
 root:=installRoot()
 addons:=chooseAddons(root)
 fmt.Println("Paket wird entpackt:",root)
 if err:=os.MkdirAll(root,0755);err!=nil{waitAndFail(err)}
 if err:=unpack(root);err!=nil{waitAndFail(err)}
 script:=filepath.Join(root,"Install.ps1")
 cmd:=exec.Command("powershell.exe","-NoProfile","-ExecutionPolicy","Bypass","-File",script,"-Addons",addons)
 cmd.Dir=root;cmd.Stdout=os.Stdout;cmd.Stderr=os.Stderr
 if err:=cmd.Run();err!=nil{waitAndFail(err)}
 fmt.Println("Addon-Update installiert. Starte ForeverCore ...")
 runtime:=exec.Command(filepath.Join(root,"ForeverCore.exe"))
 runtime.Dir=root
 if err:=runtime.Start();err!=nil{waitAndFail(err)}
 _=runtime.Process.Release()
 fmt.Println("Fertig. Jetzt WoW starten und im Chat /fc eingeben.")
 fmt.Println("Backup und Konfiguration:",root)
 fmt.Println("Enter zum Schliessen.")
 _,_=bufio.NewReader(os.Stdin).ReadString('\n')
}
